{{- /* Render hook для изображений Markdown. Если картинка — ресурс страницы (page bundle), даём её через с WebP-fallback, lazy-loading и явными width/height. Если внешний URL — простой . */ -}} {{- $dest := .Destination -}} {{- $alt := .PlainText | default "" -}} {{- $title := .Title -}} {{- $page := .Page -}} {{- /* Пытаемся найти Resource страницы по имени файла */ -}} {{- $res := "" -}} {{- with $page -}} {{- $res = .Resources.GetMatch (printf "**/%s" (path.Base $dest)) -}} {{- if not $res -}} {{- $res = .Resources.GetMatch $dest -}} {{- end -}} {{- end -}} {{- if $res -}} {{- $w := $res.Width -}} {{- $h := $res.Height -}} {{- $isRaster := in (slice "image/jpeg" "image/png" "image/webp") $res.MediaType.Type -}} {{- if $isRaster -}} {{- $maxW := 1200 -}} {{- $orig := $res -}} {{- if gt $w $maxW -}}{{ $orig = $res.Resize (printf "%dx" $maxW) -}}{{ end -}} {{- $webp := $orig.Resize (printf "%dx webp q85" $orig.Width) -}} {{- else -}} {{- /* SVG/GIF — без resize */ -}} {{- end -}} {{- else -}} {{- end -}}
