{{- /* Render hook для изображений Markdown. Если картинка — ресурс страницы (page bundle), даём её через с AVIF + WebP source'ами и оригиналом в как fallback. Lazy-loading и явные width/height — для CLS. Опт-ин AVIF: site.Params.renderImageAvif = true (по умолчанию off, так как не все сборки Hugo Extended корректно его кодируют — без libheif/libavif результат может быть пустой файл. Проверь на своей версии прежде чем включать.) Качество: site.Params.renderImageAvifQ = 60 (по умолчанию) site.Params.renderImageWebpQ = 85 (по умолчанию) Макс. ширина: site.Params.renderImageMaxW = 1200 (по умолчанию) Если внешний URL — простой . */ -}} {{- $dest := .Destination -}} {{- $alt := .PlainText | default "" -}} {{- $title := .Title -}} {{- $page := .Page -}} {{- /* Пытаемся найти Resource страницы по имени файла */ -}} {{- $res := "" -}} {{- with $page -}} {{- $res = .Resources.GetMatch (printf "**/%s" (path.Base $dest)) -}} {{- if not $res -}} {{- $res = .Resources.GetMatch $dest -}} {{- end -}} {{- end -}} {{- if $res -}} {{- $w := $res.Width -}} {{- $h := $res.Height -}} {{- $isRaster := in (slice "image/jpeg" "image/png" "image/webp") $res.MediaType.Type -}} {{- if $isRaster -}} {{- $maxW := site.Params.renderImageMaxW | default 1200 -}} {{- $avifQ := site.Params.renderImageAvifQ | default 60 -}} {{- $webpQ := site.Params.renderImageWebpQ | default 85 -}} {{- $useAvif := eq (site.Params.renderImageAvif | default false) true -}} {{- $orig := $res -}} {{- if gt $w $maxW -}}{{ $orig = $res.Resize (printf "%dx" $maxW) -}}{{ end -}} {{- $webp := $orig.Resize (printf "%dx webp q%d" $orig.Width $webpQ) -}} {{- $avif := "" -}} {{- if $useAvif -}} {{- $avif = $orig.Resize (printf "%dx avif q%d" $orig.Width $avifQ) -}} {{- end -}} {{- else -}} {{- /* SVG/GIF — без resize */ -}} {{- end -}} {{- else -}} {{- /* Внешний URL или файл из /static/. Для static-картинок приклеиваем cache-bust по mtime (тот же helper, что и для обложек), чтобы Telegram IV image-proxy не залипал на старой версии файла. */ -}} {{- $src := $dest -}} {{- if not (hasPrefix $dest "http") -}} {{- $src = printf "%s%s" $dest (partial "static_cache_bust.html" $dest) -}} {{- end -}} {{- end -}}
