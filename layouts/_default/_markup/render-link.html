{{- /* Render hook для inline-ссылок Markdown. External — добавляем rel="noopener noreferrer" + target=_blank. Internal — рендерим как обычно. */ -}} {{- $href := .Destination -}} {{- $isAbs := or (strings.HasPrefix $href "http://") (strings.HasPrefix $href "https://") -}} {{- $isExternal := and $isAbs (not (strings.HasPrefix $href site.BaseURL)) -}} {{ .Text | safeHTML }}
