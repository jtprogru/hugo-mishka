{{- /* Mermaid: подключается только если страница содержит mermaid-кодоблок или во фронтматтере явно указано mermaid: true. По умолчанию — CDN; можно переключить на self-host через site.Params.mermaidSrc. Тема Mermaid синхронизируется с темой сайта (light/dark). */ -}} {{- $hasMermaid := or (.Store.Get "hasMermaid") .Params.mermaid -}} {{- if $hasMermaid -}} {{- $src := site.Params.mermaidSrc | default "https://cdn.jsdelivr.net/npm/mermaid@11/dist/mermaid.esm.min.mjs" -}} {{- /* Палитра схем приезжает из mishka-ds: assets/mermaid/*.json генерятся из тех же токенов, что и остальные стили. Встроенные темы mermaid (default/dark) здесь не годятся — они рисуют схему в чужих цветах, и пост выглядит собранным из двух блогов. */ -}} {{- $light := resources.Get "mermaid/mermaid-config.json" -}} {{- $dark := resources.Get "mermaid/mermaid-config.dark.json" -}} {{- if or (not $light) (not $dark) -}} {{- errorf "не найдены конфиги mermaid из mishka-ds — запусти ./scripts/sync-ds.sh" -}} {{- end -}} {{- end -}}
