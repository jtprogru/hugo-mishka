{{- /* Навигация между постами: prev / next в текущей секции. */ -}} {{- $section := .Section -}} {{- $pages := where site.RegularPages "Section" $section -}} {{- $prev := $pages.Prev . -}} {{- $next := $pages.Next . -}} {{- if or $prev $next }}
{{- with $prev }} ← Предыдущий {{ .Title }} {{- end }}
{{- with $next }} Следующий → {{ .Title }} {{- end }}
{{- end }}
