{{- /* Cover-картинка поста. Порядок резолва image (как в PaperMod): 1. Если файл лежит в page-bundle (рядом с index.md) — используем Resource. 2. Если cover.relative = true — считаем путь относительным от поста. 3. Иначе (по умолчанию) — считаем что путь от корня сайта (`/`), т.е. файл в static/. 4. Абсолютные URL (http/https/начинается с /) — без изменений. Видимостью управляет: - .Params.cover.hidden — скрыть везде - .Params.cover.hiddenInSingle / site.Params... — скрыть только в single - .Params.cover.hiddenInList / site.Params... — скрыть в list Параметры partial: - context — текущая Page - location — "single" | "list" (по умолчанию single) */ -}} {{- $ctx := .context -}} {{- $location := .location | default "single" -}} {{- $cover := $ctx.Params.cover -}} {{- $img := "" -}} {{- $hidden := false -}} {{- with $cover -}} {{- $img = .image -}} {{- $hidden = .hidden -}} {{- if eq $location "single" -}} {{- $hidden = or $hidden .hiddenInSingle (eq site.Params.cover.hiddenInSingle true) -}} {{- else if eq $location "list" -}} {{- $hidden = or $hidden .hiddenInList (eq site.Params.cover.hiddenInList true) -}} {{- end -}} {{- end -}} {{- if and $img (not $hidden) -}} {{- $alt := $cover.alt | default $cover.caption | default $ctx.Title -}} {{- $caption := $cover.caption -}} {{- $relative := $cover.relative | default false -}} {{- $linkFullImages := $cover.linkFullImages | default false -}} {{- /* 1) Page-bundle resource (preferred — relative cover.image like "cover.png" living next to index.md) */ -}} {{- $res := $ctx.Resources.GetMatch (printf "*%s*" $img) -}} {{- /* 2) Global assets/covers/ — shared topical covers (e.g. cover.image: howto.png references assets/covers/howto.png). We try this before falling back to static/, because assets-resources can be resized + transcoded to WebP and emitted with srcset. */ -}} {{- if not $res -}} {{- with resources.Get (printf "covers/%s" $img) -}}{{ $res = . }}{{- end -}} {{- end -}} {{- if $res -}} {{- $isRaster := in (slice "image/jpeg" "image/png" "image/webp") $res.MediaType.Type -}} {{- $orig := $res -}} {{- $webp := "" -}} {{- $srcsetWebp := "" -}} {{- $srcsetOrig := "" -}} {{- if $isRaster -}} {{- /* 1600 — верхняя граница: обложка занимает до 1152 CSS-px, значит на retina нужен вариант заметно шире 1280. Исходники концепт-обложек лежат в 2100px, но полный 2× (2304) не окупается по весу. */ -}} {{- $maxW := 1600 -}} {{- if gt $orig.Width $maxW -}}{{ $orig = $res.Resize (printf "%dx" $maxW) }}{{ end -}} {{- $webp = $orig.Resize (printf "%dx webp q85" $orig.Width) -}} {{- /* Responsive srcset: 480w / 800w / 1200w / 1600w (capped at $orig.Width). We only generate entries whose width is <= the original to avoid upscaling. WebP entries are independent from the original-format ones so the browser can pick best by both size and format. */ -}} {{- $widths := slice -}} {{- range slice 480 800 1200 1600 -}} {{- if le . $orig.Width -}}{{ $widths = $widths | append . }}{{- end -}} {{- end -}} {{- $widths = $widths | append $orig.Width | uniq -}} {{- $webpEntries := slice -}} {{- $origEntries := slice -}} {{- range $widths -}} {{- $w := . -}} {{- $variant := $orig.Resize (printf "%dx q85" $w) -}} {{- $variantWebp := $orig.Resize (printf "%dx webp q85" $w) -}} {{- $origEntries = $origEntries | append (printf "%s %dw" $variant.RelPermalink $w) -}} {{- $webpEntries = $webpEntries | append (printf "%s %dw" $variantWebp.RelPermalink $w) -}} {{- end -}} {{- $srcsetOrig = delimit $origEntries ", " -}} {{- $srcsetWebp = delimit $webpEntries ", " -}} {{- end -}} {{- /* Широкоформатные обложки (концепт-обложки блога 1050×480, ≈2.19:1) нельзя кропать: у них по краям значимые элементы — заголовок сверху, подпись канала снизу. Общий пиксельный max-height срезал их на широком экране, поэтому помечаем такие классом и снимаем ограничение в CSS. Порог 1.8 — всё, что шире 16:9. Для векторных обложек размеров нет, но в этом блоге SVG-обложка всегда рисуется в широком концепт-формате. */ -}} {{- $wide := not $isRaster -}} {{- if and $isRaster (gt $orig.Height 0) -}} {{- if ge (div (float $orig.Width) (float $orig.Height)) 1.8 -}}{{ $wide = true }}{{- end -}} {{- end -}} {{- else -}} {{- /* 2) Внешний URL / абсолютный путь / relative */ -}} {{- $href := $img -}} {{- if or (strings.HasPrefix $img "http://") (strings.HasPrefix $img "https://") (strings.HasPrefix $img "/") -}} {{- /* как есть */ -}} {{- else if $relative -}} {{- $href = printf "%s%s" $ctx.RelPermalink $img -}} {{- else -}} {{- /* PaperMod-семантика: путь от корня сайта (static/) */ -}} {{- $href = printf "/%s" $img -}} {{- end -}} {{- /* Cache-bust для static-обложек — тот же helper, что и в cover_og_image.html. Нужен, чтобы XPath-правила IV-шаблона забирали URL с `?v=` и обходили залипший кеш Telegram-овского image-proxy. */ -}} {{- $href = printf "%s%s" $href (partial "static_cache_bust.html" $img) -}} {{- end -}} {{- end -}}
