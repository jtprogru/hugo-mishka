{{- /* Math — диспетчер math-рендерера. Включается: - frontmatter `math: true` - site.Params.math = true (глобально) - авто-детектом — на странице встречаются `$$..$$` или `\(..\)` Выбор рендерера через site.Params.mathRenderer: - "mathjax" (по умолчанию) — полный TeX/MathML/HTML, ~1 MB - "katex" — быстрее и легче, ~280 KB gzipped, чуть меньше LaTeX-features Сами рендереры — отдельные partials (mathjax.html / katex.html); пользователи темы могут переопределить любой из них в проекте. */ -}} {{- $enabled := or .Params.math site.Params.math -}} {{- if not $enabled -}} {{- $raw := .RawContent -}} {{- if or (strings.Contains $raw "$$") (strings.Contains $raw "\\(") -}} {{- $enabled = true -}} {{- end -}} {{- end -}} {{- if $enabled -}} {{- $renderer := site.Params.mathRenderer | default "mathjax" | lower -}} {{- if eq $renderer "katex" -}} {{- partial "katex.html" . -}} {{- else -}} {{- partial "mathjax.html" . -}} {{- end -}} {{- end -}}
