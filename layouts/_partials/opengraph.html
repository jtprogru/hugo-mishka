{{- /* OpenGraph партиал темы. Можно переопределить в проекте, положив свой `layouts/_partials/opengraph.html`. Фиксит места, где Hugo's built-in `_internal/opengraph.html` чуть-чуть «не дотягивает»: 1. og:locale — Facebook/Telegram ждут формат `ll_CC` (`ru_RU`, `en_US`). Берём `site.Language.LanguageCode` (если задан как `locale: ru-RU` в hugo.yaml) и меняем `-` на `_`. Иначе fallback на просто `site.Language.Lang` (например `ru`). 2. og:image:alt / og:image:width / og:image:height — пробрасываем из `.Params.cover.{alt, width, height}`. Полезно и для Telegram Instant View, и для соц-сетей. 3. article:author — заодно с tag/section/published — для превью с авторством. */ -}} {{- $title := .Title | default site.Title -}} {{- $desc := or .Description .Summary site.Params.description | plainify | htmlUnescape | chomp -}} {{- /* og:locale: язык+регион в стиле ll_CC, иначе только язык. */ -}} {{- $locale := site.Language.LanguageCode | default site.Language.Lang -}} {{- $locale = replace $locale "-" "_" -}} {{- if .IsPage }} {{- with .Section }} {{- end }} {{- $ISO8601 := "2006-01-02T15:04:05-07:00" -}} {{- with .PublishDate }} {{- end }} {{- with .Lastmod }} {{- end }} {{- with (.Params.author | default site.Params.author) }} {{- end }} {{- range .GetTerms "tags" | first 6 }} {{- end }} {{- else }} {{- end }} {{- /* Cover-картинка и её метрики/alt */ -}} {{- with .Params.cover }} {{- with .image }} {{- end }} {{- with .alt }} {{- end }} {{- with .width }}{{- end }} {{- with .height }}{{- end }} {{- end }}
