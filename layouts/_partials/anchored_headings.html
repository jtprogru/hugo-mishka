{{- /* Добавляет якорь-ссылку рядом с h2-h6 в контенте поста. Принимает на вход .Content и возвращает изменённую разметку. */ -}} {{- $content := . -}} {{- range $rid, $val := findRE ".*?" $content -}} {{- $id := index (findRE "id=\"[^\"]+\"" $val 1) 0 | replaceRE "id=\"([^\"]+)\"" "$1" -}} {{- $link := printf "#" $id -}} {{- $with := replaceRE "()" (printf "%s$1" $link) $val -}} {{- $content = replace $content $val $with -}} {{- end -}} {{- $content | safeHTML -}}
