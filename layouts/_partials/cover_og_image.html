{{- /* cover_og_image.html — returnable partial. Возвращает dict с описанием OG-картинки для текущей страницы: { url: string, width: int, height: int, alt: string } Если обложку (`.Params.cover.image`) удаётся резолвнуть как Hugo-ресурс (Page Bundle через `.Resources.GetMatch` или глобальный `assets/` через `resources.Get`), она автоматически ресайзится так, чтобы ширина не превышала 1200px — это рекомендованный OG-размер и одновременно «безопасный» для Telegram (далеко от лимита по мегапикселям). Если обложка лежит в `static/`, Hugo её обрабатывать не может, но мы приклеиваем `?v=` как cache-bust. Это нужно, потому что Telegram-овский IV-proxy (ivwebcontent.telegram.org) при условном GET'е получает 304 без тела и не всегда умеет восстановить тело из своего же кеша, из-за чего IV рендерит «Resource fetch failed». Меняем содержимое файла → меняется mtime → меняется query string → IV-proxy делает свежий запрос и получает 200 с актуальным телом. Если cover — внешний URL (`http(s)://...`) или файла нет в static/, URL отдаётся как есть. Возвращает пустой dict, если у страницы нет cover.image. */ -}} {{- $result := dict -}} {{- with .Params.cover -}} {{- $path := .image -}} {{- $alt := .alt | default "" -}} {{- $w := int (.width | default 0) -}} {{- $h := int (.height | default 0) -}} {{- $url := "" -}} {{- if $path -}} {{- /* 1. Пытаемся найти как Page Bundle resource */ -}} {{- $res := "" -}} {{- if $.IsPage -}} {{- with $.Resources.GetMatch $path -}}{{ $res = . }}{{- end -}} {{- end -}} {{- /* 2. Если не нашли — пробуем глобальный assets/covers/, затем assets/ (legacy / прочие случаи). */ -}} {{- if not $res -}} {{- with resources.Get (printf "covers/%s" $path) -}}{{ $res = . }}{{- end -}} {{- end -}} {{- if not $res -}} {{- with resources.Get $path -}}{{ $res = . }}{{- end -}} {{- end -}} {{- if $res -}} {{- /* Resize только для растровых: SVG/прочее Hugo не умеет процессить, .Width/.Resize на них падают. Для нерастровых image-MIME отдаём Permalink как есть, без width/height. */ -}} {{- $isRaster := and $res.MediaType (in (slice "image/jpeg" "image/png" "image/webp") $res.MediaType.Type) -}} {{- $isImage := and $res.MediaType (eq $res.MediaType.MainType "image") -}} {{- if $isRaster -}} {{- $proc := $res -}} {{- if gt $res.Width 1200 -}} {{- $proc = $res.Resize "1200x q85" -}} {{- end -}} {{- $url = $proc.Permalink -}} {{- $w = $proc.Width -}} {{- $h = $proc.Height -}} {{- else if $isImage -}} {{- /* SVG и прочие невекторно-непроцессируемые форматы в OG отдавать нельзя: ни Facebook, ни Twitter, ни Telegram, ни VK их не рендерят — карточка резервирует место и остаётся пустой. Ищем растрового соседа с тем же именем (его кладёт `make cover-rasterize`), и только если его нет — не отдаём ничего, чтобы сработал общий фолбэк ниже. */ -}} {{- $stem := strings.TrimSuffix (path.Ext $path) $path -}} {{- $raster := "" -}} {{- range slice "png" "webp" "jpg" -}} {{- if not $raster -}} {{- $cand := printf "%s.%s" $stem . -}} {{- if $.IsPage -}} {{- with $.Resources.GetMatch $cand }}{{ $raster = . }}{{ end -}} {{- end -}} {{- if not $raster -}} {{- with resources.Get (printf "covers/%s" $cand) }}{{ $raster = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $raster -}} {{- $proc := $raster -}} {{- if gt $raster.Width 1200 }}{{ $proc = $raster.Resize "1200x q85" }}{{ end -}} {{- $url = $proc.Permalink -}} {{- $w = $proc.Width -}} {{- $h = $proc.Height -}} {{- end -}} {{- else -}} {{- $url = $path | absURL -}} {{- end -}} {{- else -}} {{- /* Resource не нашли — внешний URL или файл в static/. Для static прикладываем cache-bust через общий helper. */ -}} {{- if hasPrefix $path "http" -}} {{- $url = $path -}} {{- else -}} {{- $bust := partial "static_cache_bust.html" $path -}} {{- $url = printf "%s%s" ($path | absURL) $bust -}} {{- end -}} {{- end -}} {{- if $url -}} {{- $result = dict "url" $url "width" $w "height" $h "alt" $alt -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Фолбэк для страниц без собственной обложки: главная, листинги, теги, статические страницы. Без него шаринг главной в Telegram или чат идёт вообще без картинки. Имя файла задаётся `params.cover.defaultImage` и ищется в `assets/covers/`; параметра нет — нет и og:image. */ -}} {{- if not $result.url -}} {{- with resources.Get (printf "covers/%s" (site.Params.cover.defaultImage | default "")) -}} {{- $proc := . -}} {{- if gt .Width 1200 }}{{ $proc = .Resize "1200x q85" }}{{ end -}} {{- $result = dict "url" $proc.Permalink "width" $proc.Width "height" $proc.Height "alt" site.Title -}} {{- end -}} {{- end -}} {{- return $result -}}
